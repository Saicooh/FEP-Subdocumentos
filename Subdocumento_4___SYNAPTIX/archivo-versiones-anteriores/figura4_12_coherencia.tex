  \centering
  \begin{tikzpicture}[
    bloque/.style={draw=SynNavyLight,fill=SynSurface,rounded corners=1mm,
      align=left,font=\fontsize{9}{11}\selectfont,text width=69mm,
      inner sep=2mm,minimum height=27mm},
    banda/.style={bloque,text width=148mm,fill=SynNavy,text=SynWhite,

    \node[banda] (plataforma) {\textbf{Zona 4: integración gobernada}\\
      Núcleos y procesos especializados: Spring Boot 4.1.1 y Spring Modulith 2.1.1.
      Contratos versionados, Correlation-ID, idempotencia, reintentos acotados
      y aislamiento por contraparte.};
    \node[bloque,below=6mm of plataforma.south west,anchor=north west] (contabilidad)
      {\textbf{Adaptador contable}\\Clientes y contratos / Finanzas y cobranza.
       El sistema vigente conserva autoridad sobre documentos, pagos y saldos;
       la plataforma prepara hechos y recibe resultados.};
    \node[bloque,below=6mm of plataforma.south east,anchor=north east] (acceso)
      {\textbf{Adaptador de acceso y CCTV}\\Autorizaciones, eventos y evidencia
       permitida. Se conserva la plataforma vigente y su red segregada;
       no se replica video ordinario.};
    \node[bloque,below=6mm of contabilidad.south west,anchor=north west] (autoridad)
      {\textbf{Meteorología y canal de autoridad}\\Canal oficial autorizado
       para DIRECTEMAR por operador; observaciones meteorológicas por gateway.
       VHF conserva la coordinación humana.};
    \node[bloque,below=6mm of acceso.south east,anchor=north east] (combustible)
      {\textbf{Adaptador del surtidor}\\Pulsos o datos mediante interfaz apta
       y barrera fuera de zona clasificada. La variante y clasificación se
       individualizan antes de autorizar la instalación.};
    \node[bloque,below=6mm of autoridad.south west,anchor=north west] (avisos)
      {\textbf{Adaptador de notificaciones}\\Twilio SMS, Meta WhatsApp y SES correo
       simultáneos en emergencia. Estado por destinatario, acuse, error,
       reintento y escalamiento.};
    \node[bloque,below=6mm of combustible.south east,anchor=north east] (navegacion)
      {\textbf{Adaptador de baja capacidad}\\Mensajes hacia embarcaciones:
       HTTPS, 1 KiB por mensaje y ventana de seis horas forman parte
       del contrato del canal seleccionado.};
    \node[banda,below=6mm of avisos.south west,anchor=north west] (frontera)
      {\textbf{Frontera de cada integración}\\Entrada por API Gateway y
       salida desde el adaptador autorizado. Protocolo y autenticación se
       individualizan por tercero: no se atribuye una API no comprobada.
       OCR: adaptador Textract separado. Catálogo y tasas en A4.2.};
    \draw[draw=SynNavyLight,thick] (plataforma.south) --
