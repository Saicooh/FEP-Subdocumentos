  \node[dominio,below=3mm of operacional.south west,anchor=north west] (maritima)

  \node[dominio,right=3mm of maritima.north east,anchor=north west] (escuela)
    {\textbf{Escuela náutica}\\
    Clase, nómina, asistencia y retiro.\\
    Publica cambios de clase.};
  \node[dominio,right=3mm of escuela.north east,anchor=north west] (infra)
    {\textbf{Infraestructura}\\
    Amarra, boya, medidor y activo físico.\\
    Publica asociación y condición.};
  \node[titulo,below=4mm of maritima.south west,anchor=north west] (servicios)
    {\textbf{Núcleo de Servicios}};
  \node[dominio,below=3mm of servicios.south west,anchor=north west] (varadero)
    {\textbf{Varadero y contratistas}\\
    Agenda, izaje, acreditación y faena.\\
    Publica plan y evidencia de ejecución.};
  \node[dominio,right=3mm of varadero.north east,anchor=north west] (combustible)
    {\textbf{Combustible}\\
    Despacho, operador, volumen y evidencia.\\
    Publica despacho validado.};
  \node[dominio,right=3mm of combustible.north east,anchor=north west] (ambiente)
    {\textbf{Ambiente y consumos}\\
    Series, residuos y evidencia ambiental.\\
    Publica consumo conciliado.};
  \node[titulo,below=4mm of varadero.south west,anchor=north west] (administrativo)
    {\textbf{Núcleo Administrativo}};
  \node[dominio,text width=65mm,below=3mm of administrativo.south west,

    {\textbf{Clientes y contratos}\\

    Publica cambios de maestros y vigencias.};
  \node[dominio,text width=65mm,right=6mm of clientes.north east,
    anchor=north west] (finanzas)
    {\textbf{Finanzas y cobranza}\\

  \node[banda,below=5mm of clientes.south west,anchor=north west] (auditoria)
    {\textbf{Noveno dominio: Auditoría y seguridad}\\

  \node[banda,below=5mm of auditoria.south west,anchor=north west] (contratos)

    No se accede a tablas ni clases internas de otro dominio.};
