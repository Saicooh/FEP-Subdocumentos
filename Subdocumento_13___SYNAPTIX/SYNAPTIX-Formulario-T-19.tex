\chapter*{Fichas de innovación}

\vspace{5mm}
Synaptix presenta cinco fichas, una por cada tipo del artículo 28 de las Bases Administrativas. Los campos siguen el Formulario T-19 y los siete elementos exigidos por el artículo 29 \parencite[pp.~19--20 y 64]{bases-admin}. La escala TRL se aplica a la integración tecnológica específica propuesta para Bahía Panitao \parencite{nasa-trl}.

\FichaInnovacion{1}{Servicio de Solicitud y Seguimiento de Mantenimiento de Embarcaciones}{%
  \CampoFicha{Tipo de innovación}{Producto o servicio de base tecnológica.}
  \CampoFicha{Nombre}{Servicio de Solicitud y Seguimiento de Mantenimiento de Embarcaciones.}
  \CampoFicha{Problema u oportunidad}{Las bases describen la participación de contratistas independientes y contabilizan 62 contratistas y cerca de 140 personas vinculadas a la operación \parencite{caso07}. Esta cantidad de actores plantea una oportunidad para dar seguimiento a las solicitudes entre armador, marina y contratista. Las bases no cuantifican solicitudes sin respuesta ni tiempos actuales de coordinación.}
  \CampoFicha{Tecnología o práctica}{Flujo digital asociado a una embarcación: el armador solicita trabajos; la marina revisa y deriva la solicitud; el contratista acepta o rechaza, programa y actualiza su avance. Se registran estados, responsables, fechas, observaciones y notificaciones \parencite{ibm-work-orders,ibm-work-order-statuses}.}
  \CampoFicha{Madurez y escala}{TRL 2 preliminar para la integración específica en la marina: el concepto está formulado, pero aún no se ha demostrado en el entorno previsto. Debe revisarse tras construir y probar un prototipo \parencite{nasa-trl}.}
  \CampoFicha{Fuentes APA 7}{Club Náutico Bahía Panitao S.~A. (2026), \textit{Bases técnicas del Caso 07: Puerto Deportivo}; IBM (s.~f.), \textit{Creating work orders} y \textit{Work order statuses}; Manning (2023), \textit{Technology readiness levels}, NASA.}
  \CampoFicha{Inserción en arquitectura}{Extensión relacionada con VAR, CTR, AUT, ALE y ADM. Validar su alcance e interfaces con la arquitectura y el T-12 para evitar duplicar funciones comprometidas.}
  \CampoFicha{Paquetes de la EDT}{[PENDIENTE: códigos para diseño, desarrollo, integración, pruebas, capacitación y piloto.]}
  \CampoFicha{Mes de materialización}{[PENDIENTE: definir con el cronograma base.]}
  \CampoFicha{Inversión requerida}{[PENDIENTE: estimar diseño, desarrollo, integración, pruebas y puesta en marcha en UF.]}
  \CampoFicha{Costo operacional}{[PENDIENTE: estimar soporte y operación incremental en UF/mes.]}
  \CampoFicha{Beneficio y cuantificación}{Reducir horas de coordinación y facilitar la detección de solicitudes pendientes o detenidas. Estimar el beneficio con las horas ahorradas y el costo horario. Incorporar inversión y costos operacionales incrementales al flujo de caja; los montos quedan pendientes de evaluación.}
  \CampoFicha{Indicador, línea base y meta}{Mediana de horas entre envío y primera respuesta del contratista. Línea base: medir antes del piloto. Meta preliminar: reducirla en 20\,\% respecto de la línea base, sujeta a validación.}
  \CampoFicha{Momento de medición}{Levantar línea base antes del piloto y medir durante el piloto y al cierre. Meses por definir según cronograma.}
  \CampoFicha{Riesgo, probabilidad e impacto}{Armadores o contratistas podrían no actualizar las solicitudes. Probabilidad e impacto: [PENDIENTE: evaluar en la matriz de riesgos.]}
  \CampoFicha{Mitigación}{Acordar responsables y plazos de actualización; capacitar a los participantes y notificar cambios de estado.}
  \CampoFicha{Contingencia}{Si el contratista no usa el sistema, la marina registrará la respuesta recibida, con autor, fecha y canal de origen.}
}

\FichaInnovacion{3}{Lectura inteligente de documentos para el ingreso asistido de antecedentes}{%

\CampoFicha{Tipo de innovación}{%
Tecnológica: OCR y extracción asistida de datos documentales.
}

\CampoFicha{Nombre}{%
Lectura inteligente de documentos para el ingreso asistido de antecedentes.
}

\CampoFicha{Problema u oportunidad}{%
Reducir la transcripción de pólizas de embarcaciones mediante campos precompletados y revisión humana, complementando RF-ADM-01 y RF-NAV-11.
}

\CampoFicha{Tecnología o práctica}{%
Amazon Textract y reglas para extraer póliza, asegurado, aseguradora, nave y vigencias. Revisión obligatoria antes de aprobar; no acredita autenticidad ni cobertura.
}

\CampoFicha{Madurez y escala}{%
TRL 2 preliminar para la integración local: concepto definido, pendiente de implementación y validación con pólizas reales.
}

\CampoFicha{Fuentes APA 7}{%
AWS: extracción documental \parencite{in3-aws-analisis,in3-aws-practicas,in3-aws-limites}; ISO: riesgos de IA \parencite{in3-iso23894}; NASA: escala TRL \parencite{in3-nasa-trl}.
}

\CampoFicha{Inserción en arquitectura}{%
AUT recibe documentos; ADM revisa y aprueba; NAV utiliza los datos aprobados. Adaptador asíncrono conectado al repositorio y la auditoría existentes.
}

\CampoFicha{Paquetes de la EDT}{%
Levantamiento, integración, interfaz de revisión, pruebas, capacitación y piloto. Códigos pendientes de conciliación con la EDT.
}

\CampoFicha{Mes de materialización}{%
Por definir en el cronograma, una vez disponibles la recepción documental, los expedientes y la revisión.
}

\CampoFicha{Inversión requerida}{%
Integración, interfaz, pruebas y capacitación. Monto en UF pendiente de estimación.
}

\CampoFicha{Costo operacional}{%
Procesamiento, almacenamiento y soporte técnico. Monto periódico en UF por estimar según volumen documental.
}

\CampoFicha{Beneficio y cuantificación}{%
Horas recuperadas frente al ingreso manual de la solución base, por costo horario, menos costo operacional incremental. Solo constituye ahorro de caja si reduce gastos efectivos.
}

\CampoFicha{Indicador, línea base y meta}{%
Reducción de la mediana del tiempo total de ingreso y revisión, incluidas correcciones. Base: ingreso manual comparable. Meta preliminar: 30\,\%, sin aumentar errores; trazabilidad del 100\,\% de aprobaciones.
}

\CampoFicha{Momento de medición}{%
Línea base antes del piloto; seguimiento durante su ejecución y evaluación al cierre. Muestra y duración por definir.
}

\CampoFicha{Riesgo, probabilidad e impacto}{%
Riesgo global preliminar: medio. Extracción errónea, revisión insuficiente, fallos externos o bajo volumen. Probabilidad e impacto por evento pendientes de evaluación.
}

\CampoFicha{Mitigación}{%
Mostrar documento de origen, contrastar identificadores y exigir aprobación humana. Aplicar permisos, cifrado, auditoría y capacitación; ampliar solo tras validar el piloto.
}

\CampoFicha{Contingencia}{%
Mantener ingreso manual ante fallos o resultados insuficientes; ajustar el extractor y reevaluar su continuidad si no demuestra beneficio.
}
}
