     \textbf{Ambiente y consumos}\\
     Relaciona mediciones, residuos, evidencia y planes de reducción.};
  \node[nucleo,left=3mm of servicios] (operacional)
    {\textbf{Núcleo Operacional}\\[3mm]
     \textbf{Operación marítima}\\

     \textbf{Escuela náutica}\\
     Relaciona clases, alumnos, botes, monitores, retornos y retiros.\\[3mm]
     \textbf{Infraestructura}\\
     Mantiene amarras, boyas, medidores, condiciones e inspecciones.};
  \node[nucleo,right=3mm of servicios] (administrativo)
    {\textbf{Núcleo Administrativo}\\[3mm]

     \textbf{Finanzas y cobranza}\\
     Relaciona hechos facturables, conciliación, saldos y expedientes.};

    {\textbf{Auditoría y seguridad}\\
     Conserva consentimientos, habilitación de identidades y decisiones;

  \node[banda,below=4mm of seguridad] (procesos)
    {\textbf{Procesos especializados compartidos}\\

     \textbf{Telemetría}: recibe lecturas identificadas.
     \textbf{Sincronización}: concilia registros y confirmaciones.
     \textbf{Auditoría}: transporta y protege evidencia de actuación.};
  \node[banda,below=7mm of procesos] (local)
    {\textbf{Célula Operacional Local: Runtime Operacional de Borde}\\
